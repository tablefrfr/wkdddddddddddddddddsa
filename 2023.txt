        public void SpawnMultipleBots(int count, string roomToJoin, string namePrefix, string gameMode)
        {
            Log.LogInfo($"[BOT] Spawning {count} bots...");
            for (int i = 0; i < count; i++)

                string botName = $"{namePrefix}_{i}_{UnityEngine.Random.Range(1000, 9999)}";
                var bot = new BotInstance(botName, roomToJoin, gameMode, _regions);
                _bots.Add(bot);
                bot.Connect();
                Thread.Sleep(400);
            }
        }

        /// <summary>
        /// Read-only snapshot of the host AppVersion for bot connect/auth only.
        /// Never writes to PhotonNetwork / PhotonNetworkController. No polling, no cache side-effects.
        /// Priority: live PhotonNetwork.AppVersion → PhotonServerSettings → _gameVersionString → fallback.
        /// </summary>
        private static string ResolveAppVersion()
        {
            // 1) Live PUN client value (what host is actually using on the wire)
            try
            {
                string ver = Photon.Pun.PhotonNetwork.AppVersion;
                if (!string.IsNullOrWhiteSpace(ver) && !IsJunkAppVersion(ver))
                    return ver.Trim();
            }
            catch { }

            // 2) PhotonServerSettings (read-only)
            try
            {
                var s = Photon.Pun.PhotonNetwork.PhotonServerSettings;
                if (s != null && s.AppSettings != null)
                {
                    string ver = s.AppSettings.AppVersion;
                    if (!string.IsNullOrWhiteSpace(ver) && !IsJunkAppVersion(ver))
                        return ver.Trim();
                }
            }
            catch { }

            // 3) Game controller private field (property setter may force "MODDED")
            try
            {
                var inst = GorillaNetworking.PhotonNetworkController.Instance;
                if (inst != null)
                {
                    var field = inst.GetType().GetField("_gameVersionString",
                        System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance);
                    var val = field?.GetValue(inst) as string;
                    if (!string.IsNullOrWhiteSpace(val) && !IsJunkAppVersion(val))
                        return val.Trim();
                }
            }
            catch { }

            return "live1.1.1.60";
        }

        private static bool IsJunkAppVersion(string ver)
        {
            if (string.IsNullOrWhiteSpace(ver)) return true;
            ver = ver.Trim();
            // Property setter on some builds hardcodes these; ignore them
            if (ver.Equals("MODDED", StringComparison.OrdinalIgnoreCase)) return true;
            if (ver.Equals("modded", StringComparison.Ordinal)) return true;
            return false;
        }

        /// <summary>
        /// Read-only preferred region from host. Sanitizes Photon quirks like "usw/" or "usw/*".
        /// </summary>
        private static string ResolvePreferredRegion(string[] fallback)
        {
            try
            {
                if (Photon.Pun.PhotonNetwork.InRoom || Photon.Pun.PhotonNetwork.IsConnected)
                {
                    var r = Photon.Pun.PhotonNetwork.CloudRegion;
                    if (!string.IsNullOrEmpty(r))
                    {
                        r = SanitizeRegion(r);
                        if (!string.IsNullOrEmpty(r))
                            return r;
                    }
                }
            }
            catch { }
            return fallback != null && fallback.Length > 0 ? SanitizeRegion(fallback[0]) : "usw";
        }

        private static string SanitizeRegion(string region)
        {
            if (string.IsNullOrWhiteSpace(region)) return "";
            region = region.Trim().ToLowerInvariant();
            // Photon sometimes returns "usw/", "usw/*", "usw/*/"
            int slash = region.IndexOf('/');
            if (slash >= 0)
                region = region.Substring(0, slash);
            region = region.TrimEnd('*', '/', ' ');
            return region;
        }

        public void SetName(string newName)
        {
            if (string.IsNullOrEmpty(newName)) return;
            _name = newName;

        public void Connect()
        {
            _connecting = true;
            Task.Run(() => ConnectAsync());
        }

        public void JoinSpecificRoom(string room)
        {
            if (string.IsNullOrEmpty(room)) return;
            _targetRoom = room.Trim().ToUpperInvariant();
            _hopPublic = false;
            _searchingRegions = true;
            _regionIndex = 0;
            _allowCreate = false;
            _inRoom = false;

            if (_photonClient != null && _photonClient.IsConnected)
            {
                if (_photonClient.InRoom)
                    _photonClient.OpLeaveRoom(false);
                DisconnectSoft();
            }
            Connect();
        }

        public void HopPublicRoom(string gameMode)
        {
            if (!string.IsNullOrEmpty(gameMode)) _gameMode = gameMode;
            _targetRoom = "";
            _hopPublic = true;
            _searchingRegions = false;
            _allowCreate = true;
            _inRoom = false;
            if (_photonClient != null && _photonClient.IsConnected)
            {
                if (_photonClient.InRoom) _photonClient.OpLeaveRoom(false);
                DisconnectSoft();
            }
            Connect();
        }

        private void DisconnectSoft()
        {
            try
            {
                if (_photonClient != null)
                {
                    try { _photonClient.RemoveCallbackTarget(this); } catch { }
                    if (_photonClient.IsConnected) _photonClient.Disconnect();
                }
            }
            catch { }
            _photonClient = null;
            _connected = false;
            _inRoom = false;
        }

        private async Task ConnectAsync()
        {
            try
            {
                GorillaBotPlugin.Log.LogInfo($"[{_name}] Connecting...");
                var (pfid, token, nonce, ticket) = await GetAuthCredentials();
                if (string.IsNullOrEmpty(pfid))
                {
                    GorillaBotPlugin.Log.LogError($"[{_name}] Failed to get auth credentials");
                    _connecting = false;
                    return;
                }

                _savedPfid = pfid;
                _savedTicket = ticket;
                _savedNonce = nonce;

                // Snapshot host AppVersion once (read-only) — never write back to host Photon
                string appVer = ResolveAppVersion();
                GorillaBotPlugin.Log.LogInfo($"[{_name}] Bot AppVersion snapshot (read-only from host): '{appVer}'");

                _photonClient = new LoadBalancingTransport2(new VoiceLog("voice"), ConnectionProtocol.Udp);
                _photonClient.AppId = "b6612b5e-ec0e-4890-8364-6c0838178c99";
                _photonClient.AppVersion = appVer;
                _photonClient.NameServerHost = "ns.exitgames.com";
                _photonClient.AddCallbackTarget(this);

                var auth = new AuthenticationValues(Guid.NewGuid().ToString("N"));
                auth.AuthType = CustomAuthenticationType.Custom;
                auth.AddAuthParameter("username", pfid);
                auth.AddAuthParameter("token", "");
                auth.SetAuthPostData(new Dictionary<string, object>
                {
                    { "AppId", "CED21" },
                    { "AppVersion", appVer },
                    { "Ticket", ticket ?? "" },
                    { "Token", "" },
                    { "Nonce", nonce ?? "" }
                });
                _photonClient.AuthValues = auth;
                _photonClient.NickName = _name;

                // Prefer the region your local client is on so private rooms resolve
                string preferred = ResolvePreferredRegion(_regions);
                // Re-order regions: preferred first (all sanitized)
                if (_regions != null && _regions.Length > 0)
                {
                    var list = new System.Collections.Generic.List<string>();
                    foreach (var r in _regions)
                    {
                        string s = SanitizeRegion(r);
                        if (!string.IsNullOrEmpty(s) && !list.Contains(s))
                            list.Add(s);
                    }
                    if (!string.IsNullOrEmpty(preferred))
                    {
                        list.Remove(preferred);
                        list.Insert(0, preferred);
                    }
                    if (list.Count == 0) list.Add("usw");
                    _regions = list.ToArray();
                    _regionIndex = 0;
                }
                string region = _regions[0];
                _currentRegion = region;
                GorillaBotPlugin.Log.LogInfo($"[{_name}] Connecting to region master: {region} (preferred={preferred}) appVer={appVer}");
                if (!_photonClient.ConnectToRegionMaster(region))
                {
                    GorillaBotPlugin.Log.LogError($"[{_name}] ConnectToRegionMaster failed");
                    _connecting = false;
                    return;
                }

                // Do NOT call _photonClient.Service() here — this runs on a background thread and
                // races with Pump() (main thread). Photon/Voice clients are not thread-safe; servicing
                // them off the main thread causes "Specified cast is not valid" and other corruption.
                // Pump() (Update, main thread) services the client every frame and drives OnJoinedRoom.
                int timeout = 0;
                while (!_inRoom && timeout < 800)
                {
                    await Task.Delay(50);
                    timeout++;
                }

                if (!_inRoom)
                    GorillaBotPlugin.Log.LogWarning($"[{_name}] Timeout (state={_photonClient?.State})");
                _connecting = false;
            }
            catch (Exception ex)
            {
                GorillaBotPlugin.Log.LogError($"[{_name}] Connect error: {ex.Message}");
                _connecting = false;
            }
        }

        private void TryNextRegionOrCreate()
        {
            // Mirror PhotonNetworkController.AttemptToJoinSpecificRoom region walk.
            // For a specific room CODE we do NOT create — that made a parallel empty room
            // while you stayed in your private lobby (same name, different room id / visibility).
            _regionIndex++;
            if (_regionIndex < _regions.Length)
            {
                string next = _regions[_regionIndex];
                GorillaBotPlugin.Log.LogInfo($"[{_name}] Room missing here — next region {next}");
                _currentRegion = next;
                _allowCreate = false;
                ReconnectToRegion(next);
            }
            else
            {
                GorillaBotPlugin.Log.LogError(
                    $"[{_name}] Room '{_targetRoom}' not found in usw/us/eu. " +
                    "Likely AppVersion mismatch, wrong region, or private room on another build. " +
                    "Bot AppVersion is a read-only host snapshot via ResolveAppVersion(). NOT creating a new room.");
                _connecting = false;
                _allowCreate = false;
            }
        }

        private void ReconnectToRegion(string region)
        {
            try
            {
                if (_photonClient != null)
                {
                    try { _photonClient.RemoveCallbackTarget(this); } catch { }
                    if (_photonClient.IsConnected) _photonClient.Disconnect();
                }
            }
            catch { }

            _photonClient = new LoadBalancingTransport2(new VoiceLog("voice"), ConnectionProtocol.Udp);
            _photonClient.AppId = "b6612b5e-ec0e-4890-8364-6c0838178c99";
            // Read-only host AppVersion snapshot (same path as initial connect)
            string appVer = ResolveAppVersion();
            _photonClient.AppVersion = appVer;
            _photonClient.NameServerHost = "ns.exitgames.com";
            _photonClient.AddCallbackTarget(this);

            if (!string.IsNullOrEmpty(_savedPfid))
            {
                var auth = new AuthenticationValues(Guid.NewGuid().ToString("N"));
                auth.AuthType = CustomAuthenticationType.Custom;
                auth.AddAuthParameter("username", _savedPfid);
                auth.AddAuthParameter("token", "");
                auth.SetAuthPostData(new Dictionary<string, object>
                {
                    { "AppId", "CED21" },
                    { "AppVersion", appVer },
                    { "Ticket", _savedTicket ?? "" },
                    { "Token", "" },
                    { "Nonce", _savedNonce ?? "" }
                });
                _photonClient.AuthValues = auth;
            }
            _photonClient.NickName = _name;
            _currentRegion = region;
            GorillaBotPlugin.Log.LogInfo($"[{_name}] ConnectToRegionMaster({region}) appVer={appVer}");
            _photonClient.ConnectToRegionMaster(region);
        }

        }


        private void SpawnBotRig()
        {
            // Real players: PhotonNetwork.Instantiate("GorillaPrefabs/Gorilla Player Networked")
            // → VRRigSerializer binds PhotonView + PhotonVoiceView to a pooled RigContainer.
            // Bots are a separate LoadBalancingClient, so we raise the same PUN Instantiation event (202).
            _botRig = null;
            _botRigGO = null;
            try
            {
                InstantiateNetworkedPlayerPrefab();
            }
            catch (Exception ex)
            {
                GorillaBotPlugin.Log.LogError($"[{_name}] InstantiateNetworkedPlayerPrefab: {ex.Message}");
            }
        }

        /// <summary>
        /// Mimic PhotonNetwork.Instantiate("GorillaPrefabs/Gorilla Player Networked") from a raw
        /// LoadBalancingClient so every client (including host) runs VRRigSerializer and gets body + speaker.
        /// </summary>
        private void InstantiateNetworkedPlayerPrefab()
        {
            if (_photonClient == null || !_photonClient.InRoom)
            {
                GorillaBotPlugin.Log.LogWarning($"[{_name}] Cannot instantiate player prefab — not in room");
                return;
            }

            int actorNr = 0;
            try { actorNr = _photonClient.LocalPlayer?.ActorNumber ?? 0; } catch { }
            if (actorNr <= 0)
            {
                GorillaBotPlugin.Log.LogWarning($"[{_name}] Invalid actor number for view ID allocation");
                return;
            }

            // PUN view IDs: ownerActor * MAX_VIEW_IDS + localIndex (MAX_VIEW_IDS = 1000)
            const int MaxViewIds = 1000;
            int viewId = actorNr * MaxViewIds + 1;
            _botRigViewId = viewId;

            if (!_rigPositionSet) ComputeRigPosition();
            string prefab = "GorillaPrefabs/Gorilla Player Networked";
            Vector3 pos = _rigPosition;

            // "Gorilla Player Networked" contains more than one PhotonView (rig + PhotonVoiceView).
            // PUN's NetworkInstantiate sets obj.ViewID = viewIDs[i] for EVERY PhotonView on the prefab,
            // so we must send one viewID per PhotonView. Sending only one (or omitting key 4) makes the
            // host throw IndexOutOfRangeException ("Index was outside the bounds of the array").
            int viewCount = GetPrefabPhotonViewCount(prefab);
            int[] viewIDs = new int[viewCount];
            for (int i = 0; i < viewCount; i++)
                viewIDs[i] = viewId + i;

            // Same payload shape as PhotonNetwork.SendInstantiate (PUN2).
            // Note: int literals 0..7 resolve to Hashtable's byte-key indexer, matching PUN's
            // keyByteZero..keyByteSeven constants, so these land as byte keys on the wire.
            var data = new Hashtable();
            data[0] = prefab;                           // keyByteZero = prefab name
            data[1] = pos;                              // keyByteOne = position
            data[2] = Quaternion.identity;              // keyByteTwo = rotation
            // keyByteThree group omitted (0)
            data[4] = viewIDs;                          // keyByteFour = one viewID per PhotonView
            // keyByteFive instantiation data omitted
            int serverTime = 0;
            try { serverTime = PhotonNetwork.ServerTimestamp; } catch { serverTime = Environment.TickCount; }
            data[6] = serverTime;                       // keyByteSix = server time (best-effort)
            data[7] = viewIDs[0];                       // keyByteSeven = instantiationId (first viewID)

            var raiseOpts = new RaiseEventOptions
            {
                CachingOption = EventCaching.AddToRoomCache,
                Receivers = ReceiverGroup.Others
            };
            // Also deliver to self? Others is enough for host; bot doesn't run PUN serialize loop
            var sendOpts = new SendOptions { Reliability = true };

            bool ok = _photonClient.OpRaiseEvent(202 /* PunEvent.Instantiation */, data, raiseOpts, sendOpts);
            GorillaBotPlugin.Log.LogInfo(
                $"[{_name}] Raised PUN Instantiation(202) prefab='{prefab}' viewId={viewId} actor={actorNr} ok={ok}");
        }

        private void ComputeRigPosition()
        {
            try
            {
                // Hardcoded forest center. The host may be an AFK client in a different room,
                // so never anchor bots to the host's position.
                float angle = UnityEngine.Random.Range(0f, 360f) * Mathf.Deg2Rad;
                float radius = UnityEngine.Random.Range(0.5f, 2.5f);
                _rigPosition = ForestCenter + new Vector3(Mathf.Cos(angle) * radius, 0f, Mathf.Sin(angle) * radius);
                _rigYaw = 0;
                _rigPositionSet = true;
                GorillaBotPlugin.Log.LogInfo($"[{_name}] Rig position = {_rigPosition}");
            }
            catch (Exception ex)
            {
                GorillaBotPlugin.Log.LogWarning($"[{_name}] ComputeRigPosition: {ex.Message}");
            }
        }

        private void SendRigTransform()
        {
            if (_photonClient == null || !_photonClient.InRoom) return;
            if (_botRigViewId <= 0 || !_rigPositionSet) return;

            try
            {
                // Field order must match VRRig.IGorillaSerializeable.OnSerializeRead:
                // headRot, rHandPos, rHandRot, lHandPos, lHandRot, bodyPos, yaw, handSync, state, ropeIndex
                object[] viewData = new object[13];
                viewData[0] = _botRigViewId;
                viewData[1] = false;
                viewData[2] = null;
                viewData[3] = Quaternion.identity;
                viewData[4] = _rigPosition + new Vector3(0.2f, 0f, 0f);
                viewData[5] = Quaternion.identity;
                viewData[6] = _rigPosition + new Vector3(-0.2f, 0f, 0f);
                viewData[7] = Quaternion.identity;
                viewData[8] = _rigPosition;
                viewData[9] = _rigYaw;
                viewData[10] = 0;
                viewData[11] = 0;
                viewData[12] = 0;

                object[] eventData = new object[3];
                eventData[0] = Environment.TickCount;
                eventData[1] = null;
                eventData[2] = viewData;

                var opts = new RaiseEventOptions { Receivers = ReceiverGroup.Others };
                var sendOpts = new SendOptions { Reliability = false };
                _photonClient.OpRaiseEvent(201, eventData, opts, sendOpts);
            }
            catch (Exception ex)
            {
                GorillaBotPlugin.Log.LogWarning($"[{_name}] SendRigTransform: {ex.Message}");
            }
        }

        private void CreatePublicRoom()
        {
            // 4-char random room like the game
            const string chars = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";

            for (int i = 0; i < 4; i++) code[i] = chars[rng.Next(chars.Length)];

                RoomOptions = new RoomOptions
                {
                    IsVisible = true,
                    IsOpen = true,

                    CustomRoomProperties = new ExitGames.Client.Photon.Hashtable
                    {
                        { "gameMode", _gameMode }
                    },
                    CustomRoomPropertiesForLobby = new[] { "gameMode" }
                }
            };
            GorillaBotPlugin.Log.LogInfo($"[{_name}] Creating public room {roomName} mode={_gameMode}");
            _photonClient.OpCreateRoom(create);
        }

            SpawnBotRig();
        }

        private System.Collections.IEnumerator PlayMp3AsVoiceCoroutine(string path)
        {
            object clip = null;
            string fail = null;
            object req = null;
            object op = null;
            Type dhType = null;

            try
            {
                // file:/// URL — UnityWebRequestMultimedia.GetAudioClip
                string url = "file:///" + path.Replace("\\", "/");
                var uwrType = Type.GetType("UnityEngine.Networking.UnityWebRequestMultimedia, UnityEngine.UnityWebRequestAudioModule")
                           ?? Type.GetType("UnityEngine.Networking.UnityWebRequestMultimedia, UnityEngine");
                var audioTypeEnum = Type.GetType("UnityEngine.AudioType, UnityEngine.AudioModule")
                                 ?? Type.GetType("UnityEngine.AudioType, UnityEngine");
                if (uwrType == null || audioTypeEnum == null)
                {
                    fail = "Audio / WebRequestAudio modules missing";
                }
                else
                {
                    var mpeg = Enum.Parse(audioTypeEnum, "MPEG");
                    var getClip = uwrType.GetMethod("GetAudioClip", new[] { typeof(string), typeof(bool), audioTypeEnum });
                    if (getClip == null)
                        getClip = uwrType.GetMethod("GetAudioClip", new[] { typeof(string), audioTypeEnum });
                    if (getClip == null)
                    {
                        fail = "GetAudioClip not found";
                    }
                    else
                    {
                        if (getClip.GetParameters().Length == 3)
                            req = getClip.Invoke(null, new object[] { url, false, mpeg });
                        else
                            req = getClip.Invoke(null, new object[] { url, mpeg });
                        var send = req.GetType().GetMethod("SendWebRequest", Type.EmptyTypes)
                                ?? req.GetType().GetMethod("Send", Type.EmptyTypes);
                        op = send.Invoke(req, null);
                        dhType = Type.GetType("UnityEngine.Networking.DownloadHandlerAudioClip, UnityEngine.UnityWebRequestAudioModule")
                              ?? Type.GetType("UnityEngine.Networking.DownloadHandlerAudioClip, UnityEngine");
                    }
                }
            }
            catch (Exception ex)
            {
                fail = ex.Message;
            }

            if (fail != null)
            {
                GorillaBotPlugin.Log.LogWarning($"[{_name}] MP3 load error: {fail}");
                yield break;
            }

            var isDoneProp = op.GetType().GetProperty("isDone");
            while (!(bool)isDoneProp.GetValue(op))
                yield return null;

            try
            {
                var resultProp = req.GetType().GetProperty("result");
                var result = resultProp != null ? resultProp.GetValue(req)?.ToString() : null;
                if (result != null && result != "Success")
                {
                    var err = req.GetType().GetProperty("error")?.GetValue(req);
                    GorillaBotPlugin.Log.LogWarning($"[{_name}] MP3 load failed: {err}");
                    yield break;
                }

                var getContent = dhType.GetMethod("GetContent", new[] { req.GetType() });
                clip = getContent.Invoke(null, new[] { req });
                _loadedClip = clip;
                if (clip == null)
                {
                    GorillaBotPlugin.Log.LogWarning($"[{_name}] MP3 clip null");
                    yield break;
                }

                // Start transmitting clip as voice mic
                bool ok = StartVoiceFromClip(clip);
                if (ok)
                    GorillaBotPlugin.Log.LogInfo($"[{_name}] Streaming MP3 through Photon Voice mic: {path}");
                else
                    GorillaBotPlugin.Log.LogWarning($"[{_name}] MP3 loaded but voice transmit failed — check VoiceInfo / LocalVoice logs above");
            }
            catch (Exception ex)
            {
                GorillaBotPlugin.Log.LogWarning($"[{_name}] MP3→voice error: {ex.Message}");
            }
        }

        private static Type FindVoiceType(string fullName)
        {
            // Prefer the assembly that actually defines VoiceClient, then scan all loaded assemblies
            try
            {
                var t = typeof(VoiceClient).Assembly.GetType(fullName);
                if (t != null) return t;
            }
            catch { }

            foreach (var asm in AppDomain.CurrentDomain.GetAssemblies())
            {
                try
                {
                    var t = asm.GetType(fullName);
                    if (t != null) return t;
                }
                catch { }
            }

            // Last resort: name-only match
            string shortName = fullName.Contains('.') ? fullName.Substring(fullName.LastIndexOf('.') + 1) : fullName;
            foreach (var asm in AppDomain.CurrentDomain.GetAssemblies())
            {
                try
                {
                    foreach (var t in asm.GetTypes())
                    {
                        if (t.Name == shortName && (t.Namespace == null || t.Namespace.StartsWith("Photon.Voice")))
                            return t;
                    }
                }
                catch { }
            }
            return null;
        }

        private static object TryParseEnum(Type enumType, params string[] names)
        {
            if (enumType == null) return null;
            foreach (var name in names)
            {
                try
                {
                    if (Enum.IsDefined(enumType, name))
                        return Enum.Parse(enumType, name);
                    // numeric fallback if enum is int-backed and name is digits
                    if (int.TryParse(name, out int n))
                        return Enum.ToObject(enumType, n);
                }
                catch { }
            }
            // First defined value as last resort
            try
            {
                var vals = Enum.GetValues(enumType);
                if (vals != null && vals.Length > 0) return vals.GetValue(0);
            }
            catch { }
            return null;
        }

        /// <returns>true if LocalVoice was created and transmit enabled</returns>
        private bool StartVoiceFromClip(object audioClip)
        {
            if (_voiceClient == null)
            {
                SetupPhotonVoice();
                if (_voiceClient == null) return false;
            }

            // Stop previous voice
            try
            {
                if (_localVoice != null)
                {
                    try { _localVoice.RemoveSelf(); } catch { }
                    try { _localVoice.Dispose(); } catch { }
                    _localVoice = null;
                }
            }
            catch { }

            try
            {
                // Step 1: Create AudioClipWrapper from the audio clip
                Type wrapperType = Type.GetType("Photon.Voice.Unity.AudioClipWrapper, PhotonVoice.API") 
                                ?? Type.GetType("Photon.Voice.Unity.AudioClipWrapper, PhotonVoice");
                
                if (wrapperType == null)
                {
                    GorillaBotPlugin.Log.LogWarning($"[{_name}] AudioClipWrapper type not found");
                    return false;
                }

                // Create wrapper: new AudioClipWrapper(audioClip)
                object reader = Activator.CreateInstance(wrapperType, new[] { audioClip });
                
                // Set Loop = true
                var loopProp = wrapperType.GetProperty("Loop");
                if (loopProp != null && loopProp.CanWrite)
                {
                    loopProp.SetValue(reader, true);
                    GorillaBotPlugin.Log.LogInfo($"[{_name}] AudioClipWrapper Loop=true");
                }

                // Step 2: Create VoiceInfo (16kHz, 16kbps, mono)
                Type voiceInfoType = FindVoiceType("Photon.Voice.VoiceInfo");
                if (voiceInfoType == null)
                {
                    GorillaBotPlugin.Log.LogWarning($"[{_name}] VoiceInfo type not found");
                    return false;
                }

                // Get sampling rate enums
                Type samplingType = FindVoiceType("Photon.Voice.SamplingRate");
                object sampling16k = null;
                
                if (samplingType != null)
                {
                    // Try common enum names
                    foreach (var name in new[] { "Sampling16000", "SamplingRate16KHz", "Sampling16K" })
                    {
                        try
                        {
                            sampling16k = Enum.Parse(samplingType, name);
                            break;
                        }
                        catch { }
                    }
                    
                    if (sampling16k == null)
                    {
                        sampling16k = Enum.ToObject(samplingType, 16000);
                    }
                }
                else
                {
                    sampling16k = 16000;
                }

                // Get frame duration enum
                Type frameType = FindVoiceType("Photon.Voice.FrameDuration");
                object frame20ms = null;
                
                if (frameType != null)
                {
                    foreach (var name in new[] { "Frame20ms", "FrameDuration20ms" })
                    {
                        try
                        {
                            frame20ms = Enum.Parse(frameType, name);
                            break;
                        }
                        catch { }
                    }
                    
                    if (frame20ms == null)
                    {
                        frame20ms = Enum.ToObject(frameType, 20);
                    }
                }
                else
                {
                    frame20ms = 20;
                }

                // Create VoiceInfo via CreateAudioOpus
                object voiceInfo = null;
                foreach (var m in voiceInfoType.GetMethods(System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.Static))
                {
                    if (m.Name != "CreateAudioOpus") continue;
                    
                    var ps = m.GetParameters();
                    if (ps.Length < 4) continue;

                    try
                    {
                        var args = new object[ps.Length];
                        args[0] = sampling16k;   // Sampling rate: 16kHz
                        args[1] = 1;             // Channels: mono
                        args[2] = frame20ms;     // Frame: 20ms
                        args[3] = 16000;         // Bitrate: 16kbps (SAFE - below 20kbps threshold)
                        
                        // Fill remaining args with defaults
                        for (int i = 4; i < ps.Length; i++)
                        {
                            if (ps[i].HasDefaultValue)
                                args[i] = ps[i].DefaultValue;
                            else if (ps[i].ParameterType.IsValueType)
                                args[i] = Activator.CreateInstance(ps[i].ParameterType);
                            else
                                args[i] = null;
                        }

                        voiceInfo = m.Invoke(null, args);
                        if (voiceInfo != null)
                        {
                            GorillaBotPlugin.Log.LogInfo($"[{_name}] ✅ VoiceInfo created (16kHz mono 16kbps SAFE)");
                            break;
                        }
                    }
                    catch (Exception ex)
                    {
                        GorillaBotPlugin.Log.LogWarning($"[{_name}] CreateAudioOpus failed: {ex.InnerException?.Message}");
                    }
                }

                if (voiceInfo == null)
                {
                    GorillaBotPlugin.Log.LogWarning($"[{_name}] Could not create VoiceInfo");
                    return false;
                }

                // Step 3: Call CreateLocalVoiceAudioFromSource
                // Signature: CreateLocalVoiceAudioFromSource(VoiceInfo voiceInfo, IAudioDesc source, AudioSampleType sampleType, IEncoder encoder = null, int channelId = 0)
                var createMethod = _voiceClient.GetType().GetMethod("CreateLocalVoiceAudioFromSource");
                if (createMethod == null)
                {
                    GorillaBotPlugin.Log.LogWarning($"[{_name}] CreateLocalVoiceAudioFromSource method not found");
                    return false;
                }

                // Get AudioSampleType.Float
                Type audioSampleType = FindVoiceType("Photon.Voice.AudioSampleType");
                object sampleFloat = null;
                if (audioSampleType != null)
                {
                    try
                    {
                        sampleFloat = Enum.Parse(audioSampleType, "Float");
                    }
                    catch
                    {
                        sampleFloat = Enum.ToObject(audioSampleType, 1);  // Float = 1 usually
                    }
                }

                // Call: voiceClient.CreateLocalVoiceAudioFromSource(voiceInfo, reader, AudioSampleType.Float, null, 0)
                var ps2 = createMethod.GetParameters();
                var args2 = new object[ps2.Length];
                args2[0] = voiceInfo;
                args2[1] = reader;  // AudioClipWrapper
                if (ps2.Length > 2) args2[2] = sampleFloat ?? 1;  // AudioSampleType.Float
                if (ps2.Length > 3) args2[3] = null;  // encoder
                if (ps2.Length > 4) args2[4] = 0;     // channelId

                _localVoice = createMethod.Invoke(_voiceClient, args2) as LocalVoice;
                
                if (_localVoice == null)
                {
                    GorillaBotPlugin.Log.LogWarning($"[{_name}] CreateLocalVoiceAudioFromSource returned null");
                    return false;
                }

                GorillaBotPlugin.Log.LogInfo($"[{_name}] ✅ LocalVoice created successfully");

                // Step 4: Configure LocalVoice
                _localVoice.TransmitEnabled = true;
                GorillaBotPlugin.Log.LogInfo($"[{_name}] ✅ TransmitEnabled=true");

                // Disable VAD
                try
                {
                    var vdProp = _localVoice.GetType().GetProperty("VoiceDetector");
                    if (vdProp != null && vdProp.CanRead)
                    {
                        var vd = vdProp.GetValue(_localVoice);
                        if (vd != null)
                        {
                            foreach (var propName in new[] { "On", "Enabled" })
                            {
                                var p = vd.GetType().GetProperty(propName);
                                if (p != null && p.CanWrite && p.PropertyType == typeof(bool))
                                {
                                    p.SetValue(vd, false);
                                    GorillaBotPlugin.Log.LogInfo($"[{_name}] ✅ VAD {propName}=false");
                                    break;
                                }
                            }
                        }
                    }
                }
                catch (Exception vadEx)
                {
                    GorillaBotPlugin.Log.LogWarning($"[{_name}] VAD config: {vadEx.Message}");
                }

                // Set InterestGroup=0 (broadcast to all)
                try
                {
                    var igProp = _localVoice.GetType().GetProperty("InterestGroup");
                    if (igProp != null && igProp.CanWrite)
                    {
                        igProp.SetValue(_localVoice, (byte)0);
                        GorillaBotPlugin.Log.LogInfo($"[{_name}] ✅ InterestGroup=0");
                    }
                }
                catch { }

                GorillaBotPlugin.Log.LogInfo($"[{_name}] ✅✅✅ VOICE READY - Bot audio should be audible!");
                return true;
            }
            catch (Exception ex)
            {
                GorillaBotPlugin.Log.LogWarning($"[{_name}] StartVoiceFromClip error: {ex.Message}\n{ex.StackTrace}");
                return false;
            }
        }

        private static object CoerceArg(Type target, object preferred, object intFallback)
        {
            if (target == null) return preferred;
            if (preferred != null && target.IsInstanceOfType(preferred)) return preferred;
            if (target.IsEnum)
            {
                try
                {
                    if (preferred != null && preferred.GetType().IsEnum)
                        return Enum.ToObject(target, Convert.ToInt32(preferred));
                    return Enum.ToObject(target, Convert.ToInt32(intFallback));
                }
                catch { }
            }
            if (target == typeof(int))
                return Convert.ToInt32(intFallback);
            if (target == typeof(byte))
                return Convert.ToByte(intFallback);
            if (target == typeof(short))
                return Convert.ToInt16(intFallback);
            if (preferred != null && target.IsAssignableFrom(preferred.GetType()))
                return preferred;
            return intFallback;
        }

        private object BuildFloatReaderFromClip(object audioClip)
        {
            // Build a simple IAudioReader<float> that loops clip samples
            try
            {
                int samples = (int)audioClip.GetType().GetProperty("samples").GetValue(audioClip, null);
                int channels = (int)audioClip.GetType().GetProperty("channels").GetValue(audioClip, null);
                int freq = (int)audioClip.GetType().GetProperty("frequency").GetValue(audioClip, null);
                float[] data = new float[samples * channels];
                var getData = audioClip.GetType().GetMethod("GetData", new[] { typeof(float[]), typeof(int) });
                getData.Invoke(audioClip, new object[] { data, 0 });

                // Downmix to mono
                float[] mono = new float[samples];
                if (channels <= 1) Array.Copy(data, mono, samples);
                else
                {
                    for (int i = 0; i < samples; i++)
                    {
                        float s = 0;
                        for (int c = 0; c < channels; c++) s += data[i * channels + c];
                        mono[i] = s / channels;
                    }
                }
                return new LoopingFloatReader(mono, freq);
            }
            catch (Exception ex)
            {
                GorillaBotPlugin.Log.LogWarning($"[{_name}] BuildFloatReader failed: {ex.Message}");
                return null;
            }
        }

        // Minimal looping float PCM reader for Photon Voice
        private class LoopingFloatReader : IAudioReader<float>
        {
            private readonly float[] _data;
            private int _pos;
            public int Channels => 1;
            public int SamplingRate { get; }
            public string Error => null;
            public LoopingFloatReader(float[] data, int rate)
            {
                _data = data;
                SamplingRate = rate;
            }
            public bool Read(float[] buf)
            {
                if (_data == null || _data.Length == 0) return false;
                for (int i = 0; i < buf.Length; i++)
                {
                    buf[i] = _data[_pos];
                    _pos++;
                    if (_pos >= _data.Length) _pos = 0;
                }
                return true;
            }
            public void Dispose() { }
        }

        public void StopAudio()
        {
            try
            {
                if (_localVoice != null)
                {
                    try { _localVoice.TransmitEnabled = false; } catch { }
                    try { _localVoice.RemoveSelf(); } catch { }
                    try { _localVoice.Dispose(); } catch { }
                    _localVoice = null;
                    GorillaBotPlugin.Log.LogInfo($"[{_name}] Voice transmit stopped");
                }
            }
            catch { }
        }
